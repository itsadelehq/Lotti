# Lotti v1 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a native macOS voice-to-text app (Wispr Flow replacement) with push-to-talk dictation, Deepgram streaming transcription, Claude Haiku polish, and Insights.

**Architecture:** SwiftUI app with a `DictationCoordinator` orchestrating isolated components (hotkey → audio capture → transcription engine → polish → paste → persist). Transcription is behind a `TranscriptionEngine` protocol so Deepgram (v1) and on-device Parakeet (later) are interchangeable. Local data in SwiftData; secrets in Keychain.

**Tech Stack:** Swift 6, SwiftUI, AppKit interop (`NSPanel`, `NSEvent`, `CGEvent`, `NSWorkspace`, `NSSound`), `AVAudioEngine`, `URLSessionWebSocketTask`, SwiftData, XCTest, Keychain Services. Anthropic + Deepgram HTTP/WS APIs.

**Spec:** `docs/superpowers/specs/2026-09-20-lotti-design.md`

## Global Constraints

- Platform: macOS 26+, Apple Silicon (arm64). Minimum deployment target `macOS 15.0`.
- App name / bundle display name: **Lotti**. Bundle id: `com.adele.lotti`.
- Transcription default engine: **Deepgram Nova** streaming (interim results on), 16kHz linear16 PCM.
- Polish model: **`claude-haiku-4-5`** ($1/M in, $5/M out). Polish is toggleable.
- API keys stored in **Keychain** only — never in `UserDefaults`, source, or logs.
- Permissions required at runtime: **Microphone** and **Accessibility**.
- Latency target: < ~700ms from key-release to pasted text.
- Every transcription engine conforms to one `TranscriptionEngine` protocol — no UI or coordinator code references Deepgram directly.

```swift
// Data/Keychain.swift
import Foundation
import Security

enum Keychain {
    static func set(_ value: String, for key: String) throws {
        let data = Data(value.utf8)
        let query: [String: Any] = [
            kSecClass as String: kSecClassGenericPassword,
            kSecAttrAccount as String: key,
        ]
        SecItemDelete(query as CFDictionary)
        var add = query
        add[kSecValueData as String] = data
        let status = SecItemAdd(add as CFDictionary, nil)
        guard status == errSecSuccess else {
            throw NSError(domain: "Keychain", code: Int(status))
        }
    }
    static func get(_ key: String) -> String? {
        let query: [String: Any] = [
            kSecClass as String: kSecClassGenericPassword,
            kSecAttrAccount as String: key,
            kSecReturnData as String: true,
            kSecMatchLimit as String: kSecMatchLimitOne,
        ]
        var out: AnyObject?
        guard SecItemCopyMatching(query as CFDictionary, &out) == errSecSuccess,
              let data = out as? Data else { return nil }
        return String(data: data, encoding: .utf8)
    }
    static func delete(_ key: String) {
        SecItemDelete([
            kSecClass as String: kSecClassGenericPassword,
            kSecAttrAccount as String: key,
        ] as CFDictionary)
    }
}
```

```swift
// Data/Models.swift
import Foundation
import SwiftData

@Model final class Dictation {
    var id: UUID
    var createdAt: Date
    var rawText: String
    var polishedText: String
    var appBundleId: String?
    var durationSeconds: Double

    init(id: UUID = UUID(), rawText: String, polishedText: String,
         appBundleId: String? = nil, durationSeconds: Double, createdAt: Date = .now) {
        self.id = id; self.rawText = rawText; self.polishedText = polishedText
        self.appBundleId = appBundleId; self.durationSeconds = durationSeconds
        self.createdAt = createdAt
    }
    var wordCount: Int {
        polishedText.split(whereSeparator: { $0.isWhitespace }).count
    }
}
```

```swift
// Data/Store.swift
import Foundation
import SwiftData

struct Stats { let totalWords: Int; let wpm: Int; let currentStreak: Int; let longestStreak: Int }

@MainActor final class Store: ObservableObject {
    let container: ModelContainer
    var context: ModelContext { container.mainContext }

    init(inMemory: Bool = false) {
        let config = ModelConfiguration(isStoredInMemoryOnly: inMemory)
        container = try! ModelContainer(for: Dictation.self, configurations: config)
    }
    func add(_ d: Dictation) { context.insert(d); try? context.save() }
    func all() -> [Dictation] {
        (try? context.fetch(FetchDescriptor<Dictation>(
            sortBy: [SortDescriptor(\.createdAt, order: .reverse)]))) ?? []
    }
    func stats(now: Date) -> Stats {
        let items = all()
        let totalWords = items.reduce(0) { $0 + $1.wordCount }
        let totalSeconds = items.reduce(0.0) { $0 + $1.durationSeconds }
        let totalWordsForWpm = items.reduce(0) { $0 + $1.wordCount }
        let wpm = totalSeconds > 0 ? Int((Double(totalWordsForWpm) / (totalSeconds / 60)).rounded()) : 0
        let (cur, longest) = Self.streaks(items.map(\.createdAt), now: now)
        return Stats(totalWords: totalWords, wpm: wpm, currentStreak: cur, longestStreak: longest)
    }
    static func streaks(_ dates: [Date], now: Date) -> (current: Int, longest: Int) {
        let cal = Calendar.current
        let days = Set(dates.map { cal.startOfDay(for: $0) }).sorted()
        guard !days.isEmpty else { return (0, 0) }
        var longest = 1, run = 1
        for i in 1..<max(days.count, 1) where days.count > 1 {
            if cal.date(byAdding: .day, value: 1, to: days[i-1]) == days[i] { run += 1 }
            else { run = 1 }
            longest = max(longest, run)
        }
        // current streak ending today
        var current = 0
        var cursor = cal.startOfDay(for: now)
        let daySet = Set(days)
        while daySet.contains(cursor) {
            current += 1
            cursor = cal.date(byAdding: .day, value: -1, to: cursor)!
        }
        return (current, max(longest, current))
    }
}
```

```swift
// Support/WordDiff.swift
import Foundation

enum WordDiff {
    static func changedWordCount(raw: String, polished: String) -> Int {
        let a = raw.split(whereSeparator: { $0.isWhitespace }).map(String.init)
        let b = polished.split(whereSeparator: { $0.isWhitespace }).map(String.init)
        var dp = Array(0...b.count)
        for i in 1...max(a.count, 1) where !a.isEmpty {
            var prev = dp[0]; dp[0] = i
            for j in 1...max(b.count, 1) where !b.isEmpty {
                let temp = dp[j]
                dp[j] = a[i-1] == b[j-1] ? prev : min(prev, dp[j], dp[j-1]) + 1
                prev = temp
            }
        }
        if a.isEmpty { return b.count }
        if b.isEmpty { return a.count }
        return dp[b.count]
    }
}
```

```swift
// Core/TranscriptionEngine.swift
import Foundation

protocol TranscriptionEngine: AnyObject {
    var onTranscript: ((String, Bool) -> Void)? { get set }
    func start() async throws
    func send(_ pcm: Data)
    func finish() async
}
```

```swift
// Core/DeepgramEngine.swift
import Foundation

enum DeepgramParser {
    private struct Frame: Decodable {
        struct Channel: Decodable { struct Alt: Decodable { let transcript: String }
            let alternatives: [Alt] }
        let is_final: Bool?
        let channel: Channel?
    }
    static func transcript(from json: Data) -> (text: String, isFinal: Bool)? {
        guard let f = try? JSONDecoder().decode(Frame.self, from: json),
              let text = f.channel?.alternatives.first?.transcript,
              !text.isEmpty else { return nil }
        return (text, f.is_final ?? false)
    }
}

final class DeepgramEngine: TranscriptionEngine {
    var onTranscript: ((String, Bool) -> Void)?
    private var task: URLSessionWebSocketTask?
    private let apiKey: String
    init(apiKey: String) { self.apiKey = apiKey }

    func start() async throws {
        var comps = URLComponents(string: "wss://api.deepgram.com/v1/listen")!
        comps.queryItems = [
            .init(name: "model", value: "nova-2"),
            .init(name: "encoding", value: "linear16"),
            .init(name: "sample_rate", value: "16000"),
            .init(name: "interim_results", value: "true"),
            .init(name: "language", value: "en"),
        ]
        var req = URLRequest(url: comps.url!)
        req.setValue("Token \(apiKey)", forHTTPHeaderField: "Authorization")
        task = URLSession.shared.webSocketTask(with: req)
        task?.resume()
        receive()
    }
    private func receive() {
        task?.receive { [weak self] result in
            guard let self else { return }
            if case .success(let msg) = result {
                if case .string(let s) = msg, let d = s.data(using: .utf8),
                   let parsed = DeepgramParser.transcript(from: d) {
                    self.onTranscript?(parsed.text, parsed.isFinal)
                }
                self.receive()
            }
        }
    }
    func send(_ pcm: Data) { task?.send(.data(pcm)) { _ in } }
    func finish() async {
        // Deepgram closes on a zero-length frame + CloseStream message
        task?.send(.string(#"{"type":"CloseStream"}"#)) { _ in }
        task?.cancel(with: .goingAway, reason: nil)
    }
}
```

```swift
// Core/AudioCapture.swift
import AVFoundation

final class AudioCapture {
    var onPCM: ((Data) -> Void)?
    var onLevel: ((Float) -> Void)?
    private let engine = AVAudioEngine()
    private var converter: AVAudioConverter?
    private let targetFormat = AVAudioFormat(
        commonFormat: .pcmFormatInt16, sampleRate: 16000, channels: 1, interleaved: true)!

    func start() throws {
        let input = engine.inputNode
        let inFormat = input.outputFormat(forBus: 0)
        converter = AVAudioConverter(from: inFormat, to: targetFormat)
        input.installTap(onBus: 0, bufferSize: 1024, format: inFormat) { [weak self] buf, _ in
            self?.process(buf)
        }
        engine.prepare()
        try engine.start()
    }
    private func process(_ buffer: AVAudioPCMBuffer) {
        guard let converter else { return }
        let ratio = targetFormat.sampleRate / buffer.format.sampleRate
        let cap = AVAudioFrameCount(Double(buffer.frameLength) * ratio) + 64
        guard let out = AVAudioPCMBuffer(pcmFormat: targetFormat, frameCapacity: cap) else { return }
        var err: NSError?
        var fed = false
        converter.convert(to: out, error: &err) { _, status in
            if fed { status.pointee = .noDataNow; return nil }
            fed = true; status.pointee = .haveData; return buffer
        }
        if let ch = out.int16ChannelData, out.frameLength > 0 {
            let data = Data(bytes: ch[0], count: Int(out.frameLength) * 2)
            onPCM?(data)
            // RMS level for waveform
            let n = Int(out.frameLength)
            var sum: Float = 0
            for i in 0..<n { let s = Float(ch[0][i]) / 32768; sum += s * s }
            onLevel?(min(1, sqrt(sum / Float(max(n,1))) * 4))
        }
    }
    func stop() {
        engine.inputNode.removeTap(onBus: 0)
        engine.stop()
    }
}
```

```swift
// Core/HotkeyManager.swift
import AppKit
import CoreGraphics

enum HotkeyEvent { case start, stop, lockToggle, cancel }

final class HotkeyStateMachine {
    var triggerKey: CGKeyCode = 0x36 // right command
    private let space: CGKeyCode = 0x31
    private(set) var isRecording = false
    private(set) var isLocked = false

    func handle(keyCode: CGKeyCode, isDown: Bool, spaceDown: Bool) -> HotkeyEvent? {
        if keyCode == space, isDown, isRecording, !isLocked {
            isLocked = true; return .lockToggle
        }
        guard keyCode == triggerKey else { return nil }
        if isDown {
            if isRecording && isLocked { // second press ends locked session
                isRecording = false; isLocked = false; return .stop
            }
            isRecording = true; return .start
        } else { // key up
            if isLocked { return nil } // hands-free: ignore release
            if isRecording { isRecording = false; return .stop }
            return nil
        }
    }
}

final class HotkeyManager {
    var onEvent: ((HotkeyEvent) -> Void)?
    let machine = HotkeyStateMachine()
    private var tap: CFMachPort?

    func start() {
        let mask = (1 << CGEventType.keyDown.rawValue) | (1 << CGEventType.keyUp.rawValue)
        tap = CGEvent.tapCreate(tap: .cgSessionEventTap, place: .headInsertEventTap,
            options: .listenOnly, eventsOfInterest: CGEventMask(mask),
            callback: { _, type, event, refcon in
                let mgr = Unmanaged<HotkeyManager>.fromOpaque(refcon!).takeUnretainedValue()
                let code = CGKeyCode(event.getIntegerValueField(.keyboardEventKeycode))
                let isDown = (type == .keyDown)
                let spaceDown = event.flags.contains(.maskSecondaryFn) // placeholder; see note
                if let ev = mgr.machine.handle(keyCode: code, isDown: isDown, spaceDown: spaceDown) {
                    DispatchQueue.main.async { mgr.onEvent?(ev) }
                }
                return Unmanaged.passUnretained(event)
            },
            userInfo: Unmanaged.passUnretained(self).toOpaque())
        if let tap {
            let src = CFMachPortCreateRunLoopSource(nil, tap, 0)
            CFRunLoopAddSource(CFRunLoopGetCurrent(), src, .commonModes)
            CGEvent.tapEnable(tap: tap, enable: true)
        }
    }
}
```

```swift
// Core/DictationCoordinator.swift
import Foundation

@MainActor final class DictationCoordinator {
    private let store: Store
    private let makeEngine: () -> TranscriptionEngine
    private let polish: PolishService?
    private let inject: Bool
    private var engine: TranscriptionEngine?
    private var audio: AudioCapture?
    private var pill: FloatingPillController?
    private var finals: [String] = []
    private var startedAt: Date?
    private var targetApp: String?

    init(store: Store, makeEngine: @escaping () -> TranscriptionEngine,
         polish: PolishService?, inject: Bool = true,
         pill: FloatingPillController? = nil, audio: AudioCapture? = nil) {
        self.store = store; self.makeEngine = makeEngine
        self.polish = polish; self.inject = inject
        self.pill = pill; self.audio = audio
    }

    func begin() {
        finals.removeAll(); startedAt = Date()
        targetApp = inject ? TextInjector.frontmostAppBundleId() : nil
        Sounds.playStart()
        pill?.show()
        let e = makeEngine(); engine = e
        e.onTranscript = { [weak self] text, isFinal in
            Task { @MainActor in if isFinal { self?.finals.append(text) } }
        }
        Task { try? await e.start() }
        if inject {
            let cap = AudioCapture(); audio = cap
            cap.onPCM = { [weak self] d in self?.engine?.send(d) }
            cap.onLevel = { [weak self] l in self?.pill?.update(level: l) }
            try? cap.start()
        }
    }

    func end(cancelled: Bool) async {
        audio?.stop()
        await engine?.finish()
        Sounds.playEnd()
        pill?.hide()
        defer { engine = nil; audio = nil }
        if cancelled { return }
        let raw = finals.joined(separator: " ")
        guard !raw.isEmpty else { return }
        let polished = (try? await polish?.polish(raw)) ?? raw
        if inject { TextInjector.paste(polished) }
        let duration = startedAt.map { Date().timeIntervalSince($0) } ?? 0
        store.add(Dictation(rawText: raw, polishedText: polished,
            appBundleId: targetApp, durationSeconds: duration))
    }
}
```
